- Root Cause: Unhandled stream aborts leave port locked.

```javascript
class SerialRobotLink {
    constructor(callbacks = {}) {
        this.callbacks = callbacks;
        this.port = null;
        this.reader = null;
        this.keepReading = false;
        this.closedPromise = null;
        this.isConnected = false;
        this.baudRate = 115200;
        this.rxBuffer = '';
        this.isDemoMode = false;
        this.demoTimer = null;

        if ('serial' in navigator) {
            navigator.serial.addEventListener('disconnect', (e) => {
                if (e.target === this.port) {
                    this._forceDisconnectCleanup();
                }
            });
        }
    }

    async connect(baud = 115200) {
        if (!('serial' in navigator)) {
            throw new Error('WebSerial not supported in this browser. Use Chrome, Edge, or Opera.');
        }
        this.baudRate = baud;
        this.port = await navigator.serial.requestPort();
        await this.port.open({ baudRate: this.baudRate });
        this.isConnected = true;
        this.keepReading = true;
        this.closedPromise = this._readLoop();
        if (this.callbacks.onConnect) this.callbacks.onConnect(this.port);
        return true;
    }

    async _readLoop() {
        while (this.port && this.port.readable && this.keepReading) {
            const textDecoder = new TextDecoderStream();
            const readableStreamClosed = this.port.readable.pipeTo(textDecoder.writable);
            this.reader = textDecoder.readable.getReader();
            try {
                while (true) {
                    const { value, done } = await this.reader.read();
                    if (done) break;
                    if (value) this._handleChunk(value);
                }
            } catch (err) {
                console.warn('WebSerial stream terminated:', err);
                break;
            } finally {
                try { this.reader.releaseLock(); } catch (_) {}
                await readableStreamClosed.catch(() => {});
            }
        }
    }

    async sendLine(cmd) {
        const trimmed = (cmd || '').trim();
        if (!trimmed) return;
        if (this.isDemoMode) {
            if (this.callbacks.onRawLine) this.callbacks.onRawLine(`[DEMO TX] > ${trimmed}`);
            return;
        }
        if (!this.isConnected || !this.port || !this.port.writable) {
            throw new Error('Robot hardware port not open');
        }
        const encoder = new TextEncoder();
        const writer = this.port.writable.getWriter();
        try {
            await writer.write(encoder.encode(trimmed + '\n'));
            if (this.callbacks.onTx) this.callbacks.onTx(trimmed);
        } finally {
            writer.releaseLock(); // GUARANTEED RELEASE
        }
    }

    async disconnect() {
        this.keepReading = false;
        if (this.reader) await this.reader.cancel().catch(() => {});
        if (this.closedPromise) await this.closedPromise.catch(() => {});
        if (this.port) {
            await this.port.close().catch(() => {});
            this.port = null;
        }
        this._forceDisconnectCleanup();
    }

    _forceDisconnectCleanup() {
        if (this.demoTimer) { clearInterval(this.demoTimer); this.demoTimer = null; }
        this.isConnected = false;
        this.isDemoMode = false;
        if (this.callbacks.onDisconnect) this.callbacks.onDisconnect();
    }
}
```

```javascript
class SimulationController {
    constructor(callbacks = {}) {
        this.isAnimating = false;
        this.generation = 0;
        this.animTimeout = null;
        this.callbacks = callbacks;
    }

    start(pathNodes, getCoordinates) {
        if (!pathNodes || pathNodes.length < 2) return false;
        this.stop();
        this.isAnimating = true;
        this.generation++;
        const currentGen = this.generation;

        if (this.callbacks.onStart) this.callbacks.onStart();
        this._step(0, pathNodes, getCoordinates, currentGen);
        return true;
    }

    stop() {
        this.isAnimating = false;
        this.generation++;
        if (this.animTimeout) {
            clearTimeout(this.animTimeout);
            this.animTimeout = null;
        }
        if (this.callbacks.onStop) this.callbacks.onStop();
    }

    _step(idx, pathNodes, getCoordinates, currentGen) {
        if (!this.isAnimating || this.generation !== currentGen) return;
        if (idx >= pathNodes.length - 1) {
            this.animTimeout = setTimeout(() => {
                if (this.generation !== currentGen) return;
                this.stop();
                if (this.callbacks.onComplete) this.callbacks.onComplete();
            }, Config.MANEUVERS.SIM_COMPLETION_DELAY_MS);
            return;
        }

        const start = pathNodes[idx];
        const end = pathNodes[idx + 1];
        if (!start || !end) { this.stop(); return; }

        const { x: x1, y: y1 } = getCoordinates(start.cellIdx);
        const { x: x2, y: y2 } = getCoordinates(end.cellIdx);

        let angle = Math.atan2(y2 - y1, x2 - x1) * 180 / Math.PI + 90;
        if (end.fn === 'BB') angle = (angle + 180) % 360;

        const moveDuration = Math.max(end.timer || 300, Config.MANEUVERS.MIN_STEP_DURATION_MS);

        if (this.callbacks.onStep) {
            this.callbacks.onStep({ x1, y1, x2, y2, angle, moveDuration, isInitial: true, stepIdx: idx });
        }

        this.animTimeout = setTimeout(() => {
            if (!this.isAnimating || this.generation !== currentGen) return;
            if (this.callbacks.onStep) {
                this.callbacks.onStep({ x1, y1, x2, y2, angle, moveDuration, isInitial: false, stepIdx: idx });
            }
            this.animTimeout = setTimeout(() => {
                if (!this.isAnimating || this.generation !== currentGen) return;
                this._step(idx + 1, pathNodes, getCoordinates, currentGen);
            }, moveDuration);
        }, Config.MANEUVERS.SIM_ROTATION_TRANSITION_MS);
    }
}
```
